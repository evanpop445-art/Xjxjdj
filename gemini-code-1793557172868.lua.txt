local function CreateToggleButton(ParentFrame, Key, Text, Y, AccentColor, OnToggleCallback)
    local Button = Instance.new("TextButton", ParentFrame)
    Button.Size = UDim2.new(1, -6, 0, 36)
    Button.Position = UDim2.new(0, 0, 0, Y)
    Button.BackgroundColor3 = DarkTheme.ButtonOff

    local Accent = Instance.new("Frame", Button)
    Accent.Size = UDim2.new(0, 4, 1, -10)
    Accent.Position = UDim2.new(0, 5, 0, 5)
    Accent.BackgroundColor3 = AccentColor
    Instance.new("UICorner", Accent).CornerRadius = UDim.new(1, 0)

    local Label = Instance.new("TextLabel", Button)
    Label.Size = UDim2.new(1, -55, 1, 0)
    Label.Position = UDim2.new(0, 15, 0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = Text
    Label.TextColor3 = Colors.White
    Label.TextSize = 11
    Label.Font = Enum.Font.GothamMedium
    Label.TextXAlignment = Enum.TextXAlignment.Left

    local State = Instance.new("TextLabel", Button)
    State.Size = UDim2.new(0, 38, 0, 20)
    State.Position = UDim2.new(1, -44, 0.5, -10)
    State.BackgroundColor3 = Color3.fromRGB(50, 20, 28)
    State.Text = Settings[Key] and "ON" or "OFF"
    State.TextColor3 = Colors.White
    State.TextSize = 10
    State.Font = Enum.Font.GothamBold
    Instance.new("UICorner", State).CornerRadius = UDim.new(0, 6)

    local function Refresh()
        local Enabled = Settings[Key]
        State.Text = Enabled and "ON" or "OFF"
        TweenService:Create(State, TweenInfo.new(0.2), {BackgroundColor3 = Enabled and DarkTheme.ButtonOn or Color3.fromRGB(50, 20, 28)}):Play()
        TweenService:Create(Button, TweenInfo.new(0.2), {BackgroundColor3 = Enabled and DarkTheme.ButtonOn or DarkTheme.ButtonOff}):Play()
    end

    Button.MouseButton1Click:Connect(function()
        Settings[Key] = not Settings[Key]
        Refresh()
        SaveSettings()
        if OnToggleCallback then OnToggleCallback(Settings[Key], Refresh) end
    end)
    
    Refresh()
    return Button
end

local function TeleportToModel(ModelKeywords, DisplayTitle, SettingKey)
    local targetPos = nil
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") then
            local name = obj.Name:lower()
            local match = true
            for _, kw in ipairs(ModelKeywords) do
                if not name:find(kw:lower()) then
                    match = false
                    break
                end
            end
            if match and obj.PrimaryPart then
                targetPos = obj.PrimaryPart.Position
                break
            elseif match then
                local part = obj:FindFirstChildWhichIsA("BasePart")
                if part then
                    targetPos = part.Position
                    break
                end
            end
        end
    end

    if targetPos and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
        LP.Character.HumanoidRootPart.CFrame = CFrame.new(targetPos + Vector3.new(0, 5, 0))
    end
end

-- Vehicle Fly Engine
---------------------------------------------------------
local controlModule = {Forward = 0, Backward = 0, Left = 0, Right = 0, Up = 0, Down = 0}

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.W then controlModule.Forward = 1
    elseif input.KeyCode == Enum.KeyCode.S then controlModule.Backward = 1
    elseif input.KeyCode == Enum.KeyCode.A then controlModule.Left = 1
    elseif input.KeyCode == Enum.KeyCode.D then controlModule.Right = 1
    elseif input.KeyCode == Enum.KeyCode.Space then controlModule.Up = 1
    elseif input.KeyCode == Enum.KeyCode.LeftShift then controlModule.Down = 1
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.W then controlModule.Forward = 0
    elseif input.KeyCode == Enum.KeyCode.S then controlModule.Backward = 0
    elseif input.KeyCode == Enum.KeyCode.A then controlModule.Left = 0
    elseif input.KeyCode == Enum.KeyCode.D then controlModule.Right = 0
    elseif input.KeyCode == Enum.KeyCode.Space then controlModule.Up = 0
    elseif input.KeyCode == Enum.KeyCode.LeftShift then controlModule.Down = 0
    end
end)

RunService.RenderStepped:Connect(function()
    if not Settings.VehicleFlyActive then return end
    local char = LP.Character
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid or not humanoid.SeatPart then return end
    
    local vehicleModel = humanoid.SeatPart.Parent
    local primaryPart = vehicleModel:IsA("Model") and (vehicleModel.PrimaryPart or vehicleModel:FindFirstChildWhichIsA("BasePart")) or humanoid.SeatPart
    
    if primaryPart then
        primaryPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        primaryPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        
        local camCFrame = Camera.CFrame
        local moveDir = Vector3.new(
            controlModule.Right - controlModule.Left,
            controlModule.Up - controlModule.Down,
            controlModule.Forward - controlModule.Backward
        )
        
        local targetVel = (camCFrame.RightVector * moveDir.X + Vector3.new(0, moveDir.Y, 0) + camCFrame.LookVector * moveDir.Z) * Settings.VehicleFlySpeed
        primaryPart.CFrame = primaryPart.CFrame + targetVel * RunService.RenderStepped:Wait()
    end
end)
